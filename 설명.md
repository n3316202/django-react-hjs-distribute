### 2. nginx 파일 추가

    # "/static/" 도메인에 도달하면 아래 alias를 수행
    # 아래 디렉토리(서버 파일시스템)을 맵핑
    location /static/ {
      alias /home/app/web/static/; 
    }

    location /media/ {
      alias /home/app/web/media/;
    }

### 3. service backend 
        python manage.py collectstatic --no-input &&

        volumes:
      - static:/app/backend/staticfiles #dev_4
      - media:/app/backend/media
